import { Component, inject, signal } from '@angular/core';
import { BuildFreeService } from './buildfree.service';

@Component({
  selector: 'app-contact',
  template: `
    <h2>Contact us</h2>
    <form (submit)="send($event)">
      <input name="name" placeholder="Your name" required />
      <input name="email" type="email" placeholder="Email" required />
      <textarea name="message" placeholder="Message" rows="4" required></textarea>
      <button>Send</button>
      <p>{{ status() }}</p>
    </form>
  `,
  styles: `
    form { display: flex; flex-direction: column; gap: 8px; max-width: 400px; }
  `,
})
export class ContactComponent {
  private api = inject(BuildFreeService);
  status = signal('');

  send(event: Event) {
    event.preventDefault();
    const form = event.target as HTMLFormElement;
    // Each input's name must match a field key in your BuildFree form
    const fields = Object.fromEntries(new FormData(form));
    this.status.set('Sending…');
    this.api.sendForm('contact-form', fields).subscribe({
      next: (res) => {
        this.status.set(res.success ? 'Thank you! We got your message.' : (res.message ?? 'Something went wrong.'));
        if (res.success) form.reset();
      },
      error: () => this.status.set('Could not send. Please try again.'),
    });
  }
}
