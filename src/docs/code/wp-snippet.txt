define( 'BUILDFREE_BASE', 'https://YOUR-BUILDFREE-API/api/v1/public/your-project-slug' );
define( 'BUILDFREE_KEY', 'PASTE-YOUR-API-KEY' );

// Talks to BuildFree. This runs on your WordPress server, so visitors never see the key.
function buildfree_request( $path, $method = 'GET', $body = null ) {
	$args = array(
		'method'  => $method,
		'timeout' => 20,
		'headers' => array(
			'x-api-key'    => BUILDFREE_KEY,
			'Content-Type' => 'application/json',
		),
	);
	if ( $body ) {
		$args['body'] = wp_json_encode( $body );
	}
	$response = wp_remote_request( BUILDFREE_BASE . $path, $args );
	if ( is_wp_error( $response ) ) {
		return null;
	}
	return json_decode( wp_remote_retrieve_body( $response ), true );
}

// Shortcode: [buildfree_products]  (or [buildfree_products collection="my-other-collection"])
function buildfree_products_shortcode( $atts ) {
	$atts      = shortcode_atts( array( 'collection' => 'products' ), $atts );
	$cache_key = 'buildfree_' . sanitize_key( $atts['collection'] );

	// Remember the answer for 5 minutes so your site stays fast
	$result = get_transient( $cache_key );
	if ( false === $result ) {
		$result = buildfree_request( '/' . rawurlencode( $atts['collection'] ) );
		if ( $result ) {
			set_transient( $cache_key, $result, 5 * MINUTE_IN_SECONDS );
		}
	}

	if ( empty( $result['data'] ) ) {
		return '<p>No products yet.</p>';
	}

	$html = '<div style="display:grid;grid-template-columns:repeat(auto-fill,minmax(200px,1fr));gap:16px">';
	foreach ( $result['data'] as $item ) {
		$name  = isset( $item['data']['name'] ) ? $item['data']['name'] : '';
		$price = isset( $item['data']['price'] ) ? $item['data']['price'] : '';

		$html .= '<div style="border:1px solid #ddd;border-radius:10px;padding:12px">';
		if ( ! empty( $item['media'][0]['url'] ) ) {
			$html .= '<img style="width:100%;border-radius:8px" src="' . esc_url( $item['media'][0]['url'] ) . '" alt="' . esc_attr( $name ) . '">';
		}
		$html .= '<h3>' . esc_html( $name ) . '</h3><p>₹' . esc_html( $price ) . '</p></div>';
	}
	return $html . '</div>';
}
add_shortcode( 'buildfree_products', 'buildfree_products_shortcode' );

// Shortcode: [buildfree_contact_form]  (or [buildfree_contact_form form="my-other-form"])
function buildfree_contact_shortcode( $atts ) {
	$atts   = shortcode_atts( array( 'form' => 'contact-form' ), $atts );
	$notice = '';

	// Did the visitor just press Send?
	if ( isset( $_POST['buildfree_nonce'] ) && wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['buildfree_nonce'] ) ), 'buildfree_contact' ) ) {
		// The keys on the left must match the field keys in your BuildFree form
		$fields = array(
			'name'    => isset( $_POST['name'] ) ? sanitize_text_field( wp_unslash( $_POST['name'] ) ) : '',
			'email'   => isset( $_POST['email'] ) ? sanitize_email( wp_unslash( $_POST['email'] ) ) : '',
			'message' => isset( $_POST['message'] ) ? sanitize_textarea_field( wp_unslash( $_POST['message'] ) ) : '',
		);
		$result = buildfree_request( '/forms/' . rawurlencode( $atts['form'] ) . '/submit', 'POST', $fields );
		$notice = ( $result && ! empty( $result['success'] ) )
			? '<p><strong>Thank you! We got your message.</strong></p>'
			: '<p>Sorry, something went wrong. Please try again.</p>';
	}

	return $notice
		. '<form method="post" style="display:flex;flex-direction:column;gap:8px;max-width:400px">'
		. wp_nonce_field( 'buildfree_contact', 'buildfree_nonce', true, false )
		. '<input name="name" placeholder="Your name" required>'
		. '<input name="email" type="email" placeholder="Email" required>'
		. '<textarea name="message" placeholder="Message" rows="4" required></textarea>'
		. '<button type="submit">Send</button>'
		. '</form>';
}
add_shortcode( 'buildfree_contact_form', 'buildfree_contact_shortcode' );
